function(add_suite_effect target display kind source flags flags2)
    set(spec_dir "${CMAKE_BINARY_DIR}/generated/${target}")
    file(MAKE_DIRECTORY "${spec_dir}")
    file(CONFIGURE OUTPUT "${spec_dir}/EffectSpec.h" CONTENT "#pragma once\n#define FOUR_EFFECT_NAME \"${display}\"\n#define FOUR_MATCH_NAME \"4x4Tools ${target}\"\n#define FOUR_VERSION 67588\n#define FOUR_FLAGS ${flags}\n#define FOUR_FLAGS2 ${flags2}\n#define FOUR_VIDEO_KIND ${kind}\n")
    set(pipl "${target}PiPL")
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/${pipl}.rc"
        COMMAND "${CMAKE_CXX_COMPILER}" /nologo /P /TC /I "${AE_SDK_ROOT}/Headers" /I "${spec_dir}"
            "/Fi${CMAKE_BINARY_DIR}/${pipl}.rr" "${CMAKE_SOURCE_DIR}/resources/SuitePiPL.r"
        COMMAND "${AE_SDK_ROOT}/Resources/PiPLtool.exe" "${pipl}.rr" "${pipl}.rrc"
        COMMAND "${CMAKE_CXX_COMPILER}" /nologo /P /TC /DMSWindows "/Fi${CMAKE_BINARY_DIR}/${pipl}.rc" "${pipl}.rrc"
        DEPENDS resources/SuitePiPL.r "${spec_dir}/EffectSpec.h" WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" VERBATIM)
    add_library(${target} MODULE ${source} "${CMAKE_BINARY_DIR}/${pipl}.rc" resources/suite_version.rc)
    target_include_directories(${target} PRIVATE ${adobe_includes} "${spec_dir}")
    set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".aex" OUTPUT_NAME "4x4${target}")
endfunction()
